原题链接：[4968. 互质数的个数 - AcWing题库](https://www.acwing.com/problem/content/4971/)

题意：求 $1 - a^b$ 中有有多少数是 $a^b$ 的互质数

# 思路

互质数数量——欧拉函数

![[欧拉函数#概念]]

但是考虑到 $a^b$ 的表示虽然可以用快速幂求，但是取模后，其欧拉函数数值会与原来不同

发现$a^b$ 的质因数与 $a$ 完全一致，只是每个多个幂指数b
即   $a^b的欧拉函数 = a^{b-1}  \times a 的欧拉函数$

快速幂求 $a^{b-1}$  再求 a的欧拉函数

# 实现

```C++
#include <iostream>
#include <algorithm>
#include <cstring>

using namespace std;
typedef long long LL;

const int mod = 998244353;



LL ans;
LL a, b;

int main()
{
    cin >> a >> b;
    
    //特判1，因为此题a取不到
    if(a == 1)
    {
        cout << 0 << endl;
        return 0;
    }
    
    LL t = a;
    ans = a;
    //cout << ans << endl;
    for(int i = 2; i <= a / i; i ++)
    {
        if(a % i == 0)
        {
            //找到素数
            ans = ans / i * (i - 1);
            while(a % i == 0)
            {
                a /= i;
            }
        }
    }
    //a大于根号a的那个数刚好是质数
    if(a > 1)
        ans = ans / a * (a - 1);
    
    //a的质因数就是a^b的质因数
    //a^b 的欧拉函数等于 a^b-1  * a的欧拉函数,为什么拆开a，是因为取模后求欧拉函数
    //不一定可以整除，数值不一定对
    
    //a的快速幂
    LL res = 1;
    b = b - 1;
    while(b)
    {
        if(b & 1) res = res * t % mod;
        b >>= 1;
        t = t * t % mod;
    }
    //cout << ans << endl;
    ans = res * ans % mod;
    
    cout << ans << endl;
    
    return 0;
}
```
